use dignity_sdk::{DignityClient, Keypair, OrderSide, DignityConfig};

#[tokio::main]
async fn main() -> Result<(), Box> {
    // 1. Initialize institutional L1 client with post-quantum Dilithium-3 keypair
    let keypair = Keypair::from_secure_enclave("0xVAL_ZURICH_FREEZONE_SWISS_01")?;
    let client = DignityClient::connect("https://gold.unykorn.org/rpc", keypair).await?;

    // 2. Query Proof of Reserves for allocated LBMA bar
    let bar_passport = client.get_bar_passport("VAL-CH-994820").await?;
    println!("Allocated Bar: {} | Fine Oz: {} | Vault: {}", 
        bar_passport.serial, bar_passport.fine_weight_oz, bar_passport.depository);

    // 3. Submit an atomic Delivery-versus-Payment (DvP) order
    let dvp_order = client.submit_dvp_order(
        OrderSide::Buy,
        "DIGAU.PHYS/USD",
        50.0,    // Quantity in Fine Troy Ounces
        2650.50, // Limit Price in USD
    ).await?;

    println!("DvP Order Committed: #{} | Block: {}", dvp_order.order_id, dvp_order.block_height);
    Ok(())
}
